import os, json, glob, hashlib, time
from pathlib import Path

ROOT = Path(__file__).resolve().parent
CELLS_DIR = ROOT / "output_json" / "cells"
OUT_DIR = ROOT / "output_json"
BRAIN_PATH = OUT_DIR / "brain.jsonl"
REPORT_PATH = ROOT / "BOOTSTRAP" / "proof" / "assemble_report.json"
MANIFEST_PATH = ROOT / "BOOTSTRAP" / "manifests" / "assemble_manifest_sha256.txt"

def sha256_file(path: Path, chunk=1024*1024) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        while True:
            b = f.read(chunk)
            if not b: break
            h.update(b)
    return h.hexdigest()

def ensure_dirs():
    (ROOT / "BOOTSTRAP" / "proof").mkdir(parents=True, exist_ok=True)
    (ROOT / "BOOTSTRAP" / "manifests").mkdir(parents=True, exist_ok=True)
    OUT_DIR.mkdir(parents=True, exist_ok=True)

def iter_json_files():
    # Accept .json and .jsonl chunks (some pipelines use either)
    patterns = [
        str(CELLS_DIR / "**" / "*.json"),
        str(CELLS_DIR / "**" / "*.jsonl"),
    ]
    files = []
    for pat in patterns:
        files.extend(glob.glob(pat, recursive=True))
    # Sort for stable, deterministic output
    files = sorted(set(files), key=lambda p: p.lower())
    return [Path(p) for p in files]

def parse_json_any(text: str, src: str):
    text = text.strip()
    if not text:
        return []
    # If it's JSONL (multiple lines), parse each line
    if "\n" in text:
        out = []
        for i, line in enumerate(text.splitlines(), 1):
            line = line.strip()
            if not line:
                continue
            try:
                out.append(json.loads(line))
            except Exception as e:
                raise ValueError(f"JSONL parse error in {src} line {i}: {e}")
        return out
    # Single JSON object
    try:
        return [json.loads(text)]
    except Exception as e:
        raise ValueError(f"JSON parse error in {src}: {e}")

def normalize_record(rec: dict, source_file: Path):
    # We do NOT assume schema—just guarantee traceability.
    # If record already has provenance fields, keep them.
    if not isinstance(rec, dict):
        rec = {"_value": rec}

    rec.setdefault("_provenance", {})
    prov = rec["_provenance"] if isinstance(rec["_provenance"], dict) else {"_provenance_raw": rec["_provenance"]}

    prov.setdefault("source_file", str(source_file))
    prov.setdefault("source_rel", str(source_file.relative_to(ROOT)) if str(source_file).startswith(str(ROOT)) else str(source_file))
    prov.setdefault("source_book_folder", source_file.parent.name)

    rec["_provenance"] = prov
    return rec

def main():
    ensure_dirs()

    if not CELLS_DIR.exists():
        raise SystemExit(f"CELLS DIR NOT FOUND: {CELLS_DIR}")

    files = iter_json_files()
    if not files:
        raise SystemExit(f"No chunk files found under: {CELLS_DIR}")

    # Build
    t0 = time.time()
    total_files = 0
    total_records = 0
    bad_files = 0
    errors = []

    # Write deterministically
    with BRAIN_PATH.open("w", encoding="utf-8") as out:
        for fp in files:
            total_files += 1
            try:
                txt = fp.read_text(encoding="utf-8", errors="replace")
                recs = parse_json_any(txt, str(fp))
                for r in recs:
                    r = normalize_record(r, fp)
                    out.write(json.dumps(r, ensure_ascii=False) + "\n")
                    total_records += 1
            except Exception as e:
                bad_files += 1
                errors.append({"file": str(fp), "error": str(e)})

    dur = round(time.time() - t0, 3)
    brain_bytes = BRAIN_PATH.stat().st_size
    brain_hash = sha256_file(BRAIN_PATH)

    # Manifest (hash all chunk files and the brain)
    with MANIFEST_PATH.open("w", encoding="ascii") as mf:
        for fp in files:
            try:
                h = sha256_file(fp)
                mf.write(f"{h}  {fp}\n")
            except Exception as e:
                mf.write(f"ERROR  {fp}  {e}\n")
        mf.write(f"{brain_hash}  {BRAIN_PATH}\n")

    report = {
        "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),
        "cells_dir": str(CELLS_DIR),
        "chunk_files_found": len(files),
        "chunk_files_processed": total_files,
        "chunk_files_failed": bad_files,
        "records_written": total_records,
        "brain_path": str(BRAIN_PATH),
        "brain_bytes": brain_bytes,
        "brain_sha256": brain_hash,
        "duration_seconds": dur,
        "errors_first_25": errors[:25],
        "note": "If chunk_files_failed > 0, see errors_first_25 and inspect those files."
    }

    REPORT_PATH.write_text(json.dumps(report, indent=2), encoding="utf-8")

    print("ASSEMBLE COMPLETE")
    print(f"Chunk files found:   {len(files)}")
    print(f"Chunk files failed:  {bad_files}")
    print(f"Records written:     {total_records}")
    print(f"Brain bytes:         {brain_bytes}")
    print(f"Brain SHA256:        {brain_hash}")
    print(f"Report:              {REPORT_PATH}")
    print(f"Manifest:            {MANIFEST_PATH}")

if __name__ == "__main__":
    main()

import json
import glob
import hashlib
import time
import sys
from pathlib import Path

# Force UTF-8 output on Windows terminals (best-effort)
try:
    sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    sys.stderr.reconfigure(encoding="utf-8", errors="replace")
except Exception:
    pass

ROOT = Path(__file__).resolve().parent
CELLS_DIR = ROOT / "output_json" / "cells"
OUT_DIR = ROOT / "output_json"
BRAIN_PATH = OUT_DIR / "brain.jsonl"

PROOF_DIR = ROOT / "BOOTSTRAP" / "proof"
MANIFEST_DIR = ROOT / "BOOTSTRAP" / "manifests"
REPORT_PATH = PROOF_DIR / "assemble_report.json"
MANIFEST_PATH = MANIFEST_DIR / "assemble_manifest_sha256.txt"
ERROR_LOG_PATH = PROOF_DIR / "assemble_errors.txt"

# ASCII control chars that can break downstream tools:
# keep \t \n \r, remove the rest in 0x00-0x1F and 0x7F
def strip_bad_controls(s: str) -> str:
    if not isinstance(s, str):
        return s
    out = []
    for ch in s:
        o = ord(ch)
        if ch in ("\t", "\n", "\r"):
            out.append(ch)
        elif (0 <= o < 32) or (o == 127):
            # drop it
            continue
        else:
            out.append(ch)
    return "".join(out)

def deep_sanitize(obj):
    # Recursively sanitize strings inside dict/list
    if isinstance(obj, str):
        return strip_bad_controls(obj)
    if isinstance(obj, list):
        return [deep_sanitize(x) for x in obj]
    if isinstance(obj, dict):
        return {k: deep_sanitize(v) for k, v in obj.items()}
    return obj

def sha256_file(path: Path, chunk=1024 * 1024) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        while True:
            b = f.read(chunk)
            if not b:
                break
            h.update(b)
    return h.hexdigest()

def ensure_dirs():
    PROOF_DIR.mkdir(parents=True, exist_ok=True)
    MANIFEST_DIR.mkdir(parents=True, exist_ok=True)
    OUT_DIR.mkdir(parents=True, exist_ok=True)

def iter_json_files():
    patterns = [
        str(CELLS_DIR / "**" / "*.json"),
        str(CELLS_DIR / "**" / "*.jsonl"),
    ]
    files = []
    for pat in patterns:
        files.extend(glob.glob(pat, recursive=True))
    files = sorted(set(files), key=lambda p: p.lower())
    return [Path(p) for p in files]

def parse_json_any(text: str, src: str):
    text = text.strip()
    if not text:
        return []

    # If it's JSONL: multiple lines with JSON objects
    if "\n" in text:
        out = []
        for i, line in enumerate(text.splitlines(), 1):
            line = line.strip()
            if not line:
                continue
            try:
                out.append(json.loads(line))
            except Exception as e:
                raise ValueError(f"JSONL parse error in {src} line {i}: {e}")
        return out

    # Single JSON object
    try:
        return [json.loads(text)]
    except Exception as e:
        raise ValueError(f"JSON parse error in {src}: {e}")

def normalize_record(rec, source_file: Path):
    if not isinstance(rec, dict):
        rec = {"_value": rec}

    # Attach provenance (non-destructive)
    rec.setdefault("_provenance", {})
    if not isinstance(rec["_provenance"], dict):
        rec["_provenance"] = {"_provenance_raw": rec["_provenance"]}

    rec["_provenance"].setdefault("source_file", str(source_file))
    try:
        rec["_provenance"].setdefault("source_rel", str(source_file.relative_to(ROOT)))
    except Exception:
        rec["_provenance"].setdefault("source_rel", str(source_file))
    rec["_provenance"].setdefault("source_book_folder", source_file.parent.name)

    # Sanitize all strings (including text fields)
    rec = deep_sanitize(rec)
    return rec

def main():
    ensure_dirs()

    if not CELLS_DIR.exists():
        raise SystemExit(f"CELLS DIR NOT FOUND: {CELLS_DIR}")

    files = iter_json_files()
    if not files:
        raise SystemExit(f"No chunk files found under: {CELLS_DIR}")

    t0 = time.time()
    total_files = 0
    total_records = 0
    bad_files = 0
    errors = []

    # Write brain.jsonl in UTF-8, replacing any weird output errors safely
    with BRAIN_PATH.open("w", encoding="utf-8", errors="replace", newline="\n") as out:
        for fp in files:
            total_files += 1
            try:
                # Read chunk file forgivingly
                txt = fp.read_text(encoding="utf-8", errors="replace")
                recs = parse_json_any(txt, str(fp))
                for r in recs:
                    r = normalize_record(r, fp)
                    out.write(json.dumps(r, ensure_ascii=False) + "\n")
                    total_records += 1
            except Exception as e:
                bad_files += 1
                errors.append({"file": str(fp), "error": str(e)})

    dur = round(time.time() - t0, 3)
    brain_bytes = BRAIN_PATH.stat().st_size
    brain_hash = sha256_file(BRAIN_PATH)

    # Manifest (chunk hashes + brain hash)
    with MANIFEST_PATH.open("w", encoding="utf-8", errors="replace", newline="\n") as mf:
        for fp in files:
            try:
                h = sha256_file(fp)
                mf.write(f"{h}  {fp}\n")
            except Exception as e:
                mf.write(f"ERROR  {fp}  {e}\n")
        mf.write(f"{brain_hash}  {BRAIN_PATH}\n")

    # Error log (human-readable)
    with ERROR_LOG_PATH.open("w", encoding="utf-8", errors="replace", newline="\n") as ef:
        for item in errors:
            ef.write(f"{item['file']} :: {item['error']}\n")

    report = {
        "timestamp": time.strftime("%Y-%m-%d %H:%M:%S"),
        "cells_dir": str(CELLS_DIR),
        "chunk_files_found": len(files),
        "chunk_files_processed": total_files,
        "chunk_files_failed": bad_files,
        "records_written": total_records,
        "brain_path": str(BRAIN_PATH),
        "brain_bytes": brain_bytes,
        "brain_sha256": brain_hash,
        "duration_seconds": dur,
        "errors_first_25": errors[:25],
        "note": "UTF-8 hardened + control-char sanitization. See assemble_errors.txt if failures > 0."
    }

    REPORT_PATH.write_text(json.dumps(report, indent=2, ensure_ascii=False), encoding="utf-8", errors="replace")

    print("\nASSEMBLE COMPLETE")
    print(f"Chunk files found:   {len(files)}")
    print(f"Chunk files failed:  {bad_files}")
    print(f"Records written:     {total_records}")
    print(f"Brain bytes:         {brain_bytes}")
    print(f"Brain SHA256:        {brain_hash}")
    print(f"Report:              {REPORT_PATH}")
    print(f"Errors:              {ERROR_LOG_PATH}")
    print(f"Manifest:            {MANIFEST_PATH}\n")

if __name__ == "__main__":
    main()
