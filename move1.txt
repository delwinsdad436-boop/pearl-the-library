param(
  [string]$BrainPath = "output_json\brain.jsonl",
  [string]$ManifestPath = "BOOTSTRAP\manifests\manifest_sha256.txt"
)

if (!(Test-Path $BrainPath)) { throw "No brain file: $BrainPath" }
if (!(Test-Path $ManifestPath)) { throw "No manifest file: $ManifestPath" }

$ts = (Get-Date).ToString("yyyyMMdd-HHmmss")
$runId = "run-" + $ts

$brainHash = (Get-FileHash $BrainPath -Algorithm SHA256).Hash
$manifestHash = (Get-FileHash $ManifestPath -Algorithm SHA256).Hash
$lines = (Get-Content $BrainPath -ReadCount 0).Count
$size = (Get-Item $BrainPath).Length

$packet = [ordered]@{
  run_id = $runId
  timestamp = (Get-Date).ToString("yyyy-MM-dd HH:mm:ss")
  brain = [ordered]@{
    path = (Resolve-Path $BrainPath).Path
    sha256 = $brainHash
    lines = $lines
    bytes = $size
  }
  manifest = [ordered]@{
    path = (Resolve-Path $ManifestPath).Path
    sha256 = $manifestHash
  }
}

$packet | ConvertTo-Json -Depth 6 | Set-Content -Encoding UTF8 "BOOTSTRAP\proof\commit_packet.json"
